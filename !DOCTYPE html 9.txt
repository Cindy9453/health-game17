<script>

let hero=100, monster=100;
let used=new Set();

/* ===== 100題 ===== */
const qs=[
{q:"睡眠不足影響？",a:["記憶下降","變強","長高","健康"],c:0},
{q:"運動好處？",a:["健康","肥胖","疲勞","失眠"],c:0},
{q:"蛋白質？",a:["修復","糖","油","酒"],c:0}
];

while(qs.length<100){
  let i=qs.length+1;
  qs.push({
    q:`第${i}題：健康行為？`,
    a:["運動","熬夜","抽菸","暴食"],
    c:0
  });
}

/* 題目 */
function getQ(){
  if(used.size===100){
    alert("破關");
    location.reload();
  }
  let i;
  do{i=Math.floor(Math.random()*100);}while(used.has(i));
  used.add(i);
  return qs[i];
}

/* 血條 */
function update(){
  document.getElementById("heroHP").style.width=hero+"%";
  document.getElementById("monsterHP").style.width=monster+"%";
}

/* ⭐ 子彈（完全修正版） */
function shoot(fromHero){

  let b=document.createElement("div");
  b.className="bullet";
  document.body.appendChild(b);

  let x=fromHero?200:window.innerWidth-200;
  let y=window.innerHeight/2;

  b.style.left=x+"px";
  b.style.top=y+"px";

  let move=setInterval(()=>{

    x += fromHero ? 20 : -20;
    b.style.left=x+"px";

    if(x>window.innerWidth-200 || x<200){
      clearInterval(move);
      b.remove();

      if(fromHero) monster-=10;
      else hero-=10;

      update();
      check();
    }

  },16);
}

/* 題目 */
function load(){
  let q=getQ();
  document.getElementById("q").innerText=q.q;

  let box=document.getElementById("ans");
  box.innerHTML="";

  q.a.forEach((t,i)=>{
    let btn=document.createElement("button");
    btn.innerText=t;
    btn.onclick=()=>{
      if(i===q.c) shoot(true);
      else shoot(false);
      load();
    };
    box.appendChild(btn);
  });
}

/* 勝負 */
function check(){
  if(hero<=0) alert("你輸了");
  if(monster<=0) alert("你贏了");
}

/* start */
update();
load();

</script>
